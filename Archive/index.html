archive.json — Entry Structure
Each entry in archive.json is an object in the root array.
{
  "id":     "arc001",               // unique string, used in URL refs
  "type":   "research|drama|analysis|artifact",
  "title":  "Full title of the entry",
  "author": "handle or name",
  "date":   "YYYY-MM-DD",
  "tags":   ["tag1", "tag2"],       // lowercase, hyphenated
  "status": "archived|active|developing",
  "tldr":   "One-paragraph summary shown on the card",
  "blocks": [ ...block objects... ] // content body
}
Text Blocks
{ "type": "text", "format": "paragraph",  "content": "..." }
{ "type": "text", "format": "header",     "content": "Section title" }
{ "type": "text", "format": "subheader",  "content": "Sub-section" }
{ "type": "text", "format": "list",       "items": ["item1", "item2"] }
{ "type": "text", "format": "quote",      "content": "..." }
{ "type": "text", "format": "caption",    "content": "Small grey text" }
Media Blocks
// Image (click to lightbox)
{ "type": "image", "src": "path/or/url.png", "caption": "optional" }

// Embed — YouTube, iframes, etc.
{ "type": "embed", "src": "https://youtube.com/embed/...", "height": 360, "caption": "optional" }

// Code block with syntax label
{ "type": "code", "language": "python", "content": "print('hello')" }
Tutorial / Guide Blocks
// Numbered tutorial steps — each step can have nested blocks
{
  "type": "steps",
  "steps": [
    {
      "title": "Install the tool",
      "content": "Optional description under the step title.",
      "blocks": [
        { "type": "code", "language": "bash", "content": "npm install -g sometool" },
        { "type": "callout", "level": "info", "content": "Requires Node 18+." }
      ]
    },
    {
      "title": "Configure",
      "content": "Open config.json and set your values."
    }
  ]
}

// Keyboard shortcut display
// combo: true = keys shown with + between them (Ctrl+S)
// combo: false = keys shown with 'then' (sequential)
{ "type": "kbd", "label": "Save", "keys": ["Ctrl", "S"], "combo": true, "description": "Save file" }
{ "type": "kbd", "label": "Search and replace", "keys": ["Ctrl", "H"], "combo": true }

// Checklist — items can be strings or {text, done} objects
{
  "type": "checklist",
  "items": [
    { "text": "Read the prerequisites", "done": true },
    { "text": "Install dependencies", "done": true },
    { "text": "Configure environment", "done": false },
    "Optional plain string item"
  ]
}

// Diff — before/after code comparison
// Lines starting with "- " are red (removed), "+ " are green (added), rest is context
{
  "type": "diff",
  "label": "config.js — before and after",
  "lines": [
    "  const port = process.env.PORT;",
    "- app.listen(3000);",
    "+ app.listen(port || 3000);",
    "  module.exports = app;"
  ]
}
Info & Structure Blocks
// Callout — levels: info, warning, critical
{ "type": "callout", "level": "warning", "content": "..." }

// Spoiler / collapsible section (accepts nested blocks)
{
  "type": "spoiler",
  "title": "Click to expand — contains spoilers",
  "blocks": [ ...any blocks... ]
}

// Table
{
  "type": "table",
  "headers": ["Column A", "Column B", "Column C"],
  "rows": [
    ["Row 1 A", "Row 1 B", "Row 1 C"],
    ["Row 2 A", "Row 2 B", "Row 2 C"]
  ],
  "caption": "optional table caption"
}

// Stat cards (for research summaries, metrics)
{
  "type": "stats",
  "items": [
    { "value": "38M", "label": "Files archived" },
    { "value": "640GB", "label": "Total size" },
    { "value": "2009", "label": "Year captured" }
  ]
}

// Redacted block
{ "type": "redacted", "reason": "optional reason text" }
Research / Drama Blocks
// Cited source with optional Wayback Machine archive URL
{
  "type": "source",
  "label": "Source label",
  "url": "https://...",
  "archived": "https://web.archive.org/...",
  "date": "YYYY-MM-DD",
  "note": "Context about why this source matters"
}

// Timeline of events
{
  "type": "timeline",
  "events": [
    { "date": "2024-01", "label": "Event name", "detail": "What happened." },
    { "date": "2024-03", "label": "Response",   "detail": "What followed." }
  ]
}

// Two-column layout (any block type per column)
{
  "type": "columns",
  "columns": [
    { "type": "text", "format": "paragraph", "content": "Left side." },
    { "type": "text", "format": "paragraph", "content": "Right side." }
  ]
}

// Labeled evidence box (nested blocks inside a labeled frame)
{
  "type": "evidence",
  "label": "Exhibit A — Original tweet screenshot",
  "blocks": [
    { "type": "image", "src": "screenshot.png" },
    { "type": "text", "format": "caption", "content": "Captured Oct 14 2024, since deleted." }
  ]
}
nanoposts.json — Nanopost Structure

  "id":      "nano001",
  "date":    "2026-03-05T14:22:00",   // ISO 8601
  "content": "The text of the post.",
  "tags":    ["tag1", "tag2"],         // optional
  "ref": {                             // optional — links to archive entry
    "id":    "arc001",
    "label": "Entry title for display"
  },
  "link": {                            // optional — external link
    "url":   "https://...",
    "label": "Link label"
  }
}
